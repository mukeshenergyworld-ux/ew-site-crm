  var K = function (c) { return String(c || "").replace(/\s+/g, "").toUpperCase(); };
  grid.forEach(function (r) { var c = K(r[1]); if (c) have[c] = 1; var n = Number(r[0]); if (!isNaN(n) && n > maxSno) maxSno = n; });
  var fo = DriveApp.getFoldersByName("EW catalogue pictures - TOTO"), folder = fo.hasNext() ? fo.next() : DriveApp.createFolder("EW catalogue pictures - TOTO");
  var base = "https://mukeshenergyworld-ux.github.io/ew-site-crm/assets/";
  var grab = function (url) {
    try {
      var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: true, headers: { "User-Agent": "Mozilla/5.0" } });
      var b = res.getBlob(), ct = String(b.getContentType() || "");
      if (res.getResponseCode() === 200 && /^image\//.test(ct) && b.getBytes().length > 3000) return b;
    } catch (e) { }
    return null;
  };
  var out = [], specs = [], log = [], src = {};
  TOTO_N13_.forEach(function (r) {
    var code = K(r.code);
    if (have[code]) { log.push(code + ": already there, skipped"); return; }
    var order = r.first === "file" ? [["file", r.asset ? base + r.asset : ""], ["web", r.web]] : [["web", r.web], ["file", r.asset ? base + r.asset : ""]];
    var pic = "", used = "";
    for (var k = 0; k < order.length && !pic; k++) {
      if (!order[k][1]) continue;
      var b = grab(order[k][1]);
      if (b) {
        var nm = "TOTO_" + code.replace(/[#\/]/g, "_") + "_n13" + (order[k][0] === "web" ? "w" : "") + ".jpg";
        var f = folder.createFile(b.setName(nm));
        pic = "https://drive.google.com/thumbnail?id=" + f.getId() + "&sz=w800"; used = order[k][0] + " " + b.getBytes().length;
      }
    }
    maxSno++; have[code] = 1; src[code] = { from: r.src, pic: used, web: r.web };
    out.push([maxSno, code, r.family, r.desc, r.family, "Per Pc.", Number(r.price) || "", "", "TOTO", "", pic]);
    specs.push([r.specs || ""]);
    log.push(code + ": added, " + (pic ? "picture " + used : "NO PICTURE") + ", Rs " + r.price);
  });
  if (out.length) {
    sh.getRange(last + 1, 1, out.length, 11).setValues(out);
    sh.getRange(last + 1, 12, specs.length, 1).setValues(specs);
    SpreadsheetApp.flush(); teamBust_();
    var au = teamSheet_("audit"), hd = au.getRange(1, 1, 1, Math.max(au.getLastColumn(), 1)).getValues()[0];
    var rec = { id: "CA-" + Date.now(), createdAt: new Date().toISOString(), actor: "Mukesh Verma", action: "catalogAppend", target: "TOTO",
      detail: JSON.stringify({ n: out.length, files: "Abhi Dewan x7, Roshan ji, Sanjay Kumar ji, Product Display, CP Architect", rows: src }).slice(0, 49000), ip: "" };
    au.appendRow(hd.map(function (k) { return rec[k] === undefined ? "" : rec[k]; }));
  }
  Logger.log(out.length + " added\n" + log.join("\n"));
  return log;
}
